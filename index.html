const fireworksLayer = document.getElementById('fireworksLayer');
const sparkColors = ['#ec4899', '#a855f7', '#f472b6', '#c084fc', '#fbbf24', '#f9a8d4'];

function spawnFireworkBurst(cx, cy) {
const count = 22;
for (let i = 0; i < count; i++) {
const angle = (Math.PI * 2 * i) / count + (Math.random() - 0.5) * 0.25;
const dist = 52 + Math.random() * 58;
const tx = Math.cos(angle) * dist;
const ty = Math.sin(angle) * dist;

const spark = document.createElement('span');
spark.className = 'spark';
spark.style.left = cx + 'px';
spark.style.top = cy + 'px';
spark.style.color = sparkColors[i % sparkColors.length];
spark.style.background = sparkColors[i % sparkColors.length];
spark.style.setProperty('--tx', tx.toFixed(1) + 'px');
spark.style.setProperty('--ty', ty.toFixed(1) + 'px');
fireworksLayer.appendChild(spark);

setTimeout(() => spark.remove(), 1200);
}
}

const w = window.innerWidth;

}

function spawnConfetti(count = 26) {
const colors = ['#f472b6', '#c084fc', '#fbcfe8', '#e9d5ff', '#f9a8d4'];
for (let i = 0; i < count; i++) {
const piece = document.createElement('span');
piece.className = 'confetti-piece';
piece.style.left = (12 + Math.random() * 76) + 'vw';
piece.style.background = colors[i % colors.length];
piece.style.setProperty('--x', ((Math.random() - 0.5) * 140) + 'px');
piece.style.setProperty('--dur', (1.8 + Math.random() * 1.2) + 's');
document.body.appendChild(piece);
setTimeout(() => piece.remove(), 3100);
}
}
